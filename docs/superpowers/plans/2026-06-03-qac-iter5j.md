# qac iter-5j — per-CT evidence-list endpoint Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`).

**Goal:** `GET /api/runs/{id}/evidence` (optional `?step=`) lists a run's evidence rows (per-CT + aggregate) with payloads inlined, so the per-tray outcomes become queryable.

**Architecture:** New `store.ListEvidence(runID, stepID)` + a list handler with a payload-inlining DTO (since `Evidence.Payload` is `json:"-"`), registered next to the existing single-fetch evidence route.

**Tech Stack:** Go 1.23 (bare module `qac`), `net/http` ServeMux 1.22. Spec: `docs/superpowers/specs/2026-06-03-qac-iter5j-design.md`.

// ListEvidence returns a run's evidence rows, ordered created_at then
// deviceslot (per-CT rows grouped stably; the NULL-deviceslot aggregate
// sorts first). stepID == "" returns all rows for the run; otherwise only
// that step's. Returns an empty slice when none.
func (s *Store) ListEvidence(ctx context.Context, runID, stepID string) ([]Evidence, error) {
	q := `SELECT id, run_id, step_id, content_type, payload, created_at, deviceslot
	      FROM evidence WHERE run_id = ?`
	args := []any{runID}
	if stepID != "" {
		q += ` AND step_id = ?`
		args = append(args, stepID)
	}
	q += ` ORDER BY created_at, deviceslot`
	rows, err := s.db.QueryContext(ctx, q, args...)
	if err != nil {
		return nil, fmt.Errorf("list evidence: %w", err)
	}
	defer rows.Close()
	var out []Evidence
	for rows.Next() {
		var e Evidence
		if err := rows.Scan(&e.ID, &e.RunID, &e.StepID, &e.ContentType, &e.Payload, &e.CreatedAt, &e.Deviceslot); err != nil {
			return nil, fmt.Errorf("scan evidence: %w", err)
		}
		out = append(out, e)
	}
	return out, rows.Err()
}
```

// evidenceListItem is the list response shape. Evidence.Payload is json:"-"
// on the store struct (served raw by evidenceHandler), so the list inlines
// it as raw JSON here.
type evidenceListItem struct {
	ID          string          `json:"id"`
	StepID      string          `json:"step_id"`
	Deviceslot  *string         `json:"deviceslot,omitempty"`
	ContentType string          `json:"content_type"`
	CreatedAt   int64           `json:"created_at"`
	Payload     json.RawMessage `json:"payload"`
}

// evidenceListHandler serves GET /api/runs/{id}/evidence (optional ?step=).
func evidenceListHandler(s *store.Store) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		runID := r.PathValue("id")
		if runID == "" {
			writeError(w, http.StatusBadRequest, codeInvalidRequest, "Run id is required")
			return
		}
		rows, err := s.ListEvidence(r.Context(), runID, r.URL.Query().Get("step"))
		if err != nil {
			writeError(w, http.StatusInternalServerError, codeInternal, "Failed to list evidence")
			return
		}
		items := make([]evidenceListItem, 0, len(rows))
		for _, e := range rows {
			items = append(items, evidenceListItem{
				ID: e.ID, StepID: e.StepID, Deviceslot: e.Deviceslot,
				ContentType: e.ContentType, CreatedAt: e.CreatedAt,
				Payload: json.RawMessage(e.Payload),
			})
		}
		writeJSON(w, http.StatusOK, map[string]any{"evidence": items})
	}
}
```
