# qac iter-5i — reconcile the projections Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`).

**Goal:** Make the probe's verdict reach the operator-visible `run.status` by having the store projection (which the API serves) mirror `engine.Project`, both sharing one `verdict` logic package, with a parity test guaranteeing they can't diverge.

**Architecture:** New `internal/verdict` package (consts + `Worse` + `ToStatus`). `engine` and `rlcc_action_probe` re-point to it. The store projection's `StepCompleted` consumes `Verdict` (worst-wins via a new `worst_verdict` column), `StepFailed` pins worst=failed, `OperatorAction{signoff}`→`signed_off`. A parity test asserts store status == `engine.Project(evs).Status`.

**Tech Stack:** Go 1.23 (bare module `qac`), `modernc.org/sqlite`. Spec: `docs/superpowers/specs/2026-06-03-qac-iter5i-design.md`.

## Key facts (read first)
- `engine/event.go`: status consts `StatusPending/Running/Passed/Warning/Failed/SignedOff` (values `pending/running/passed/warning/failed/signed_off`); verdict consts `VerdictPassed/Warning/Failed` (`passed/warning/failed`).
- `engine/project.go`: `worse(a,b)` (rank `{"":0, passed:1, warning:2, failed:3}`, returns higher); `verdictToStatus(v)`; `OperatorActionPayload.Action` with `if p.Action == "signoff"` → signed_off.
- `store/events.go`: mirrors wire payloads locally (`stepCompletedPayload` etc.); `StepCompleted` does `_ = p` + bump updated_at; `StepFailed` → `UPDATE runs SET status='failed' … WHERE status != 'signed_off'`; `OperatorAction` (line 211) only bumps updated_at. store must NOT import engine (engine imports store).
- `store/migrations.go`: `addBundleColumnIfMissing` pattern — `SELECT COUNT(*) FROM pragma_table_info('runs') WHERE name='X'` then `ALTER TABLE runs ADD COLUMN X TEXT NOT NULL DEFAULT ''`.
- `getRunHandler` returns `{"run": run, "events":…, "racks":…}` — run status is nested under `run`.

```go
// Package verdict is the single source of qac's run-status vocabulary and
// the verdict→status rollup. It depends on nothing so both the store
// projection (internal/store) and the engine projection (internal/engine)
// can share it without an import cycle.
package verdict

// Verdict values an assertion probe produces (also the verdict-derived
// status strings — same literals).
const (
	Passed  = "passed"
	Warning = "warning"
	Failed  = "failed"
)

// Run status values.
const (
	StatusPending   = "pending"
	StatusRunning   = "running"
	StatusPassed    = "passed"
	StatusWarning   = "warning"
	StatusFailed    = "failed"
	StatusSignedOff = "signed_off"
)

// Worse returns the worse (higher-ranked) of two verdicts. Empty = no
// verdict yet (best). Used to roll up a run's worst step verdict.
func Worse(a, b string) string {
	rank := map[string]int{"": 0, Passed: 1, Warning: 2, Failed: 3}
	if rank[a] >= rank[b] {
		return a
	}
	return b
}

// ToStatus maps a (worst) verdict to a run status. Empty/unknown → running.
func ToStatus(v string) string {
	switch v {
	case Passed:
		return StatusPassed
	case Warning:
		return StatusWarning
	case Failed:
		return StatusFailed
	default:
		return StatusRunning
	}
}
```

```go
func (s *Store) addWorstVerdictColumnIfMissing(ctx context.Context) error {
	var n int
	if err := s.db.QueryRowContext(ctx,
		`SELECT COUNT(*) FROM pragma_table_info('runs') WHERE name = 'worst_verdict'`).Scan(&n); err != nil {
		return fmt.Errorf("introspect worst_verdict column: %w", err)
	}
	if n > 0 {
		return nil
	}
	if _, err := s.db.ExecContext(ctx,
		`ALTER TABLE runs ADD COLUMN worst_verdict TEXT NOT NULL DEFAULT ''`); err != nil {
		return fmt.Errorf("add worst_verdict column: %w", err)
	}
	return nil
}
```

```go
package engine

import (
	"context"
	"encoding/json"
	"path/filepath"
	"testing"

	"qac/internal/store"
)

func TestProjectionParity_StoreMatchesEngineProject(t *testing.T) {
	scripts := map[string][]struct {
		kind    string
		payload map[string]any
	}{
		"gatherer-only": {
			{"RunCreated", map[string]any{"template_id": "t", "created_by": "x"}},
			{"StepStarted", map[string]any{"step_id": "s1"}},
			{"StepCompleted", map[string]any{"step_id": "s1"}},
		},
		"one-fail": {
			{"RunCreated", map[string]any{"template_id": "t", "created_by": "x"}},
			{"StepCompleted", map[string]any{"step_id": "s1", "verdict": "passed"}},
			{"StepCompleted", map[string]any{"step_id": "s2", "verdict": "failed"}},
		},
		"all-pass": {
			{"RunCreated", map[string]any{"template_id": "t", "created_by": "x"}},
			{"StepCompleted", map[string]any{"step_id": "s1", "verdict": "passed"}},
			{"StepCompleted", map[string]any{"step_id": "s2", "verdict": "passed"}},
		},
		"warning-then-fail": {
			{"RunCreated", map[string]any{"template_id": "t", "created_by": "x"}},
			{"StepCompleted", map[string]any{"step_id": "s1", "verdict": "warning"}},
			{"StepCompleted", map[string]any{"step_id": "s2", "verdict": "failed"}},
		},
		"signoff": {
			{"RunCreated", map[string]any{"template_id": "t", "created_by": "x"}},
			{"StepCompleted", map[string]any{"step_id": "s1", "verdict": "passed"}},
			{"OperatorAction", map[string]any{"action": "signoff"}},
		},
	}
	for name, script := range scripts {
		t.Run(name, func(t *testing.T) {
			ctx := context.Background()
			s, err := store.Open(ctx, filepath.Join(t.TempDir(), "p.db"))
			if err != nil {
				t.Fatalf("Open: %v", err)
			}
			t.Cleanup(func() { _ = s.Close() })
			runID := "01HZX0000000000000PARIT0"
			for _, e := range script {
				b, _ := json.Marshal(e.payload)
				if err := s.AppendEvent(ctx, runID, e.kind, b); err != nil {
					t.Fatalf("append %s: %v", e.kind, err)
				}
			}
			runs, _ := s.ListRuns(ctx)
			var storeStatus string
			for _, r := range runs {
				if r.ID == runID {
					storeStatus = r.Status
				}
			}
			evs, _ := s.LoadEvents(ctx, runID)
			engineStatus := Project(evs).Status
			if storeStatus != engineStatus {
				t.Errorf("%s: store status=%q, engine.Project status=%q (projections diverged)", name, storeStatus, engineStatus)
			}
		})
	}
}
```
Adjust `Project`'s signature/return to match its real shape (it returns a `RunState` with `.Status`). If `RunCreated` requires specific payload fields to project without error, mirror what other engine tests pass.
